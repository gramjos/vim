325 " Robustly run shell command, insert stdout at cursor
326 function! Qs(...) abort
327     if a:0 < 1 " an integer that holds the total count of arguments passed to the function.
328         echoerr "Provide at least 1 word as shell command"
329         return
330     endif
331     let cmd = join(a:000, ' ') " a List containing all the arguments passed to the function
332     echom cmd
333     let output = split(system(cmd), "\n")
334     if v:shell_error
335         echoerr "Shell error occurred: " . cmd
336         return
337     endif
338     call append(line('.') - 1, output)
339 endfunction
